Supabase 관리자 RLS 정책 가이드

Supabase 대시보드(SQL Editor)에서 아래 SQL을 실행하면, phiskim@gmail.com 계정에 전체 회원·시험·페이지뷰 읽기 권한이 부여됩니다. (이미 실행했다면 다시 실행해도 안전합니다.)

-- 1. phiskim@gmail.com 에 role='admin' 부여
update public.service_members sm
set role = 'admin', updated_at = now()
from auth.users u
where sm.user_id = u.id
  and lower(u.email) = 'phiskim@gmail.com'
  and sm.service = 'gram';

-- 2. 관리자는 모든 회원 조회 및 상태 변경 가능
drop policy if exists "members_select_admin" on public.service_members;
create policy "members_select_admin" on public.service_members
  for select using ((auth.jwt()->>'email' = 'phiskim@gmail.com'));

-- 3. 관리자는 전체 시험 결과 조회 가능 (퀴즈 + 모의고사)
drop policy if exists "quiz_select_admin" on gram.quiz_results;
create policy "quiz_select_admin" on gram.quiz_results
  for select using ((auth.jwt()->>'email' = 'phiskim@gmail.com'));

-- 4. 관리자는 전체 학습 로그 조회 가능
drop policy if exists "study_select_admin" on gram.study_log;
create policy "study_select_admin" on gram.study_log
  for select using ((auth.jwt()->>'email' = 'phiskim@gmail.com'));

-- 5. 화면 조회수 및 접속 시간대 통계 테이블 생성
create table if not exists gram.page_views (
  id bigint generated always as identity primary key,
  path text not null,
  page_title text,
  referrer text,
  user_id uuid references auth.users on delete set null,
  hour int not null default extract(hour from (now() at time zone 'Asia/Seoul')),
  day date not null default (current_date at time zone 'Asia/Seoul')::date,
  created_at timestamptz not null default now()
);
alter table gram.page_views enable row level security;

create policy "page_views_insert_all" on gram.page_views for insert with check (true);
create policy "page_views_select_admin" on gram.page_views for select using ((auth.jwt()->>'email' = 'phiskim@gmail.com'));

grant all on table gram.page_views to anon, authenticated, service_role;
